## 2. 目标
5 列表页 `<n-data-table>` 加 `:row-props` 使每行支持 Tab 聚焦 + Enter 跳转详情。

## 3. 具体 diff（通用模板）

**5 文件**：每个文件 script 段追加 rowProps 函数 + template 绑定

### 3.1 script setup 段
```diff
+ import { useRouter } from 'vue-router'
+ const router = useRouter()

+ function rowProps(row: any) {
+   return {
+     tabindex: 0,
+     role: 'button',
+     'aria-label': `${row.name || row.id} 详情`,
+     onKeydown: (e: KeyboardEvent) => {
+       if (e.key === 'Enter') router.push(`/${getRouteBase()}/${row.id}`)
+     },
+     style: 'cursor: pointer',
+     onClick: () => router.push(`/${getRouteBase()}/${row.id}`),
+   }
+ }
+
+ function getRouteBase() {
+   // 5 文件分别写：'offer' / 'demand' / 'onboarding' / 'interview' / 'candidate'
+   return 'offer'  // 各文件填实际
+ }
```

### 3.2 template 段
```diff
- <n-data-table :columns="columns" :data="dataSource" />
+ <n-data-table :columns="columns" :data="dataSource" :row-props="rowProps" />
```
