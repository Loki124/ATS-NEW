.n-data-table .n-data-table-th {
  background: rgba(255, 255, 255, .5);
  backdrop-filter: blur(8px);
  -webkit-backdrop-filter: blur(8px);

  color: var(--ink-soft);
}

  background: rgba(255, 255, 255, .06);
  color: var(--ink);
