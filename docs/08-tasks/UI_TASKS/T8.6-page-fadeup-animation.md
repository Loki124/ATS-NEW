## 3. 具体 diff（5 文件，每个文件改 1 行）

```css
/* v2: 入场动效（与 Dashboard 一致） */

  animation: wb-fade-up var(--duration-slow) var(--ease-out) both;
```

```bash
  const anim = await page.evaluate(() => {
    const c = document.querySelector('.page-container')
    if (!c) return null
    const cs = getComputedStyle(c)
    return { name: cs.animationName, duration: cs.animationDuration }
  })
  console.log(`${route}:`, JSON.stringify(anim))
  // 期望: { name: 'wb-fade-up', duration: '0.32s' 或 '0.4s' }
  await page.close()
}
```
